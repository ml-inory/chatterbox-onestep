cmake_minimum_required(VERSION 3.15)
project(chatterbox_s3gen_onestep_sdk LANGUAGES CXX C)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_library(chatterbox_s3gen_onestep_sdk STATIC src/model_runner.cpp)
target_include_directories(chatterbox_s3gen_onestep_sdk PUBLIC include)
add_executable(model_example examples/main.cpp)
add_executable(openai_server examples/openai_server.cpp)
add_executable(openai_client examples/openai_client.cpp)
if(AX_RUNTIME_ROOT)
  target_include_directories(chatterbox_s3gen_onestep_sdk PUBLIC ${AX_RUNTIME_ROOT}/include)
  target_link_directories(chatterbox_s3gen_onestep_sdk PUBLIC ${AX_RUNTIME_ROOT}/lib)
  target_link_libraries(chatterbox_s3gen_onestep_sdk PUBLIC ax_engine ax_sys)
endif()
target_link_libraries(model_example PRIVATE chatterbox_s3gen_onestep_sdk)
target_link_libraries(openai_server PRIVATE chatterbox_s3gen_onestep_sdk)
target_link_libraries(openai_client PRIVATE)
target_link_libraries(openai_server PRIVATE pthread)
# DSP 数值循环需要 -O3 + fast-math（默认无优化时板端 Griffin-Lim 会慢到分钟级）
target_compile_options(openai_server PRIVATE -O3 -ffast-math)
if(AX_RUNTIME_ROOT)
  # BSP 共享库的部分 AX_NPU_* 符号由板端驱动/运行库提供，交叉链接时放行未定义符号
  target_link_options(model_example PRIVATE "-Wl,--allow-shlib-undefined")
  target_link_options(openai_server PRIVATE "-Wl,--allow-shlib-undefined")
endif()
